\begin{algorithmic}[1]
\Procedure{FlushRemaining}{$b, G, \iota$}
    \If{$b = \varepsilon$} \State \textbf{return} \EndIf
    \State $(\tau, \_) \gets \Call{Parse}{G, b}$
    \ForAll{$s \in \Call{TopLevelStatements}{\tau}$}
        \State $c \gets b[\,s.\mathit{start}\,..\,s.\mathit{end}\,]$
        \State $o \gets \iota.\Call{Execute}{c}$
        \State \textbf{yield} $o$
    \EndFor
\EndProcedure
\end{algorithmic}
